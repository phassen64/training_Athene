}

Function Get-LargeFiles2 ($location = "C:\Windows", $length = 1MB, [switch]$subdir)
{
	$schalter=$False
	If($subdir)
	{
		$schalter=$True
	}
	Get-ChildItem $location -Recurse:$schalter | Where-Object 
	{$_.length -ge $length}
}

Filter NewFiles ([int]$tage)
{
	$jetzt = Get-Date
	$_ | Where-Object { ($jetzt - $_.LastWriteTime).Days –lt $tage}
}
